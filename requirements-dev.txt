# Development/test dependencies. Not required to run MuxCls itself.
#
# The floor is the oldest version the suite is actually run against, not a
# guess: 9.0.3 is what local runs use, and CI resolves this to the newest 9.x
# on every run, so both ends of the range get exercised.
pytest>=9.0.3

# Lint, typecheck and coverage. CI runs all three; the coverage number is
# reported rather than gated on a threshold.
pytest-cov>=7.1.0

# The linters are pinned to a compatible release, not a floor. A new ruff adds
# rules to its default set, so an open range breaks CI on code nobody touched -
# which is exactly what happened when these were first added. Bump these
# deliberately, run them locally, then push.
ruff~=0.15.14
mypy~=2.1.0
